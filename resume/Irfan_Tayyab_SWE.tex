% --- Section headings: \section*{Title} --------------------------------
%  Bold grey title with a full-width rule underneath. The zero-width strut
%  gives every heading the same depth, so the rule sits at the same distance
%  whether or not the title has descenders (g, p, y).
\titleformat{\section}
  {\sectionfont}                                          % format
  {}{0pt}                                                 % no numbers
  {\headingtitle}                                         % title
  [\vspace{\rulegap}{\color{black}\titlerule[\ruleheight]}] % rule below
\newcommand{\headingtitle}[1]{#1\rule[-3pt]{0pt}{3pt}}
\titlespacing*{\section}{0pt}{\sectionbefore}{\sectionafter}

% --- Bullet lists: plain \begin{itemize} ... \end{itemize} ------------

  label      = \textendash,                              % small dash
  leftmargin = \dimexpr\entryindent+\bulletindent\relax,   % lines up under the job title
  rightmargin= \bulletright,                             % shorter lines, easier to read
  labelwidth = \bulletindent,
  labelsep   = 0pt,
  align      = left,
  topsep     = 0pt,
  itemsep    = \itemgap,                     % space between bullets
  parsep     = 0pt,
  partopsep  = 0pt,
  before     = \vspace{\bulletgap},         % job title -> first bullet
  after      = \endgraf\addvspace{\jobgap}, % last bullet -> next job
}

% \job{Role}{Company}{Dates}
%  One line: a grey marker bar, then "Role, Company" on the left and the
%  dates flush right. The title starts \entryindent in; bullets line up with it.

  \makebox[\entryindent][l]{\jobmarker}%
  {\semiboldfont\color{black}#1}, #2\hfill #3\par
}

% skills list: "Label: items" as running text, the label bold and the items
%  following it on the same line, wrapped lines starting at the margin. A
%  group's wrapped lines are \skilllinegap further apart than body text, and
%  groups \skillgap.
\newenvironment{skills}
  {\par\setlength{\baselineskip}{\dimexpr\baselineskip+\skilllinegap\relax}%
   \let\skillbefore\relax}
  {\par}
% \skill{Label}{items} — one group in the skills list, a paragraph of its own.
\newcommand{\skill}[2]{%
  \skillbefore
  {\bfseries #1:} #2\par
  \def\skillbefore{\vspace{\dimexpr\skillgap-\skilllinegap\relax}}%
}

\newcommand{\project}[2]{\textbullet\ \href{#1}{\uline{#2\ \extarrow}}}
