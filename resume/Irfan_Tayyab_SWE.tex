% --- Section headings: \section*{Title} --------------------------------
%  Bold grey title with a full-width rule underneath. The zero-width strut
%  gives every heading the same depth, so the rule sits at the same distance
%  whether or not the title has descenders (g, p, y).
\titleformat{\section}
  {\sectionfont}                                          % format
  {}{0pt}                                                 % no numbers
  {\headingtitle}                                         % title
  [\vspace{\rulegap}{\color{black}\titlerule[\ruleheight]}] % rule below
\newcommand{\headingtitle}[1]{#1\rule[-3pt]{0pt}{3pt}}
\titlespacing*{\section}{0pt}{\sectionbefore}{\sectionafter}

  parsep     = 0pt,
  partopsep  = 0pt,
